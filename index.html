// 將您取得的 Google API 網址貼在這裡 const GOOGLE_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbwMHrbgGvy23Q4WTj_OzMN68ibCdwiG9YbA66AGndwzNRq_YMLSSlMi7UoXp63oo-Ui3w/exec"; // 1. 從 Google 試算表讀取最新即時資產數據 async function fetchFromGoogleSheet() { try { const res = await fetch(GOOGLE_SCRIPT_URL); const data = await res.json(); // 自動帶入試算表算好的即時市值與佔比 document.getElementById('etfValue').value = data.etfValue; document.getElementById('cashValue').value = data.cashValue; updatePortfolio(); // 更新網頁圖表與評估 alert("已成功從 Google 試算表同步最新股價與資產數據！"); } catch (err) { console.error("同步失敗", err); } } // 2. 當在 GitHub 網頁新增買賣紀錄時，同步寫入 Google 試算表 async function sendTradeToGoogleSheet(tradeData) { // tradeData 格式範例: { type: '買進', shares: 1000, price: 150, fee: 142 } try { await fetch(GOOGLE_SCRIPT_URL, { method: 'POST', body: JSON.stringify(tradeData) }); alert("交易紀錄已成功同步儲存至 Google 試算表！"); } catch (err) { console.error("寫入失敗", err); } }
